<script>
const produtos = [
  { 
    nome: "Camiseta personalizada", 
    preco: 49.99,
    img: "https://via.placeholder.com/150"
  },
  { 
    nome: "Calça baggy moletom", 
    preco: 79.99,
    img: "https://via.placeholder.com/150"
  },
  { 
    nome: "TN sunset", 
    preco: 299.99,
    img: "tn_no_bg.png", // 🔥 SUA IMAGEM SEM FUNDO
    destaque: true
  }
];

let carrinho = {};

function renderProdutos() {
  const container = document.getElementById("produtos");

  produtos.forEach((p, i) => {
    container.innerHTML += `
      <div class="card ${p.destaque ? 'destaque' : ''}">
        <img src="${p.img}">
        <h3>${p.nome}</h3>
        <p>R$ ${p.preco.toFixed(2)}</p>
        <div class="controls">
          <button onclick="add(${i})">+</button>
          <span id="qtd-${i}">0</span>
          <button onclick="remove(${i})">-</button>
        </div>
      </div>
    `;
  });
}

function add(i) {
  carrinho[i] = (carrinho[i] || 0) + 1;
  atualizar();
}

function remove(i) {
  if (carrinho[i]) carrinho[i]--;
  atualizar();
}

function atualizar() {
  let subtotal = 0;

  produtos.forEach((p, i) => {
    let qtd = carrinho[i] || 0;
    document.getElementById(`qtd-${i}`).innerText = qtd;
    subtotal += qtd * p.preco;
  });

  let frete = 0;
  let select = document.getElementById("bairro");
  if (select.value) {
    frete = parseFloat(select.selectedOptions[0].dataset.frete);
  }

  document.getElementById("subtotal").innerText = subtotal.toFixed(2);
  document.getElementById("frete").innerText = frete.toFixed(2);
  document.getElementById("total").innerText = (subtotal + frete).toFixed(2);
}

document.getElementById("bairro").addEventListener("change", atualizar);

function finalizar() {
  let select = document.getElementById("bairro");

  if (Object.keys(carrinho).length === 0) {
    alert("Adicione produtos!");
    return;
  }

  if (!select.value) {
    alert("Selecione o bairro!");
    return;
  }

  let mensagem = "Olá, quero fazer um pedido:%0A🛍️ Produtos:%0A";

  produtos.forEach((p, i) => {
    if (carrinho[i]) {
      mensagem += `- ${p.nome} (Qtd: ${carrinho[i]})%0A`;
    }
  });

  mensagem += `🚚 Entrega: ${select.value}%0A`;
  mensagem += `💰 Total: R$ ${document.getElementById("total").innerText}`;

  window.open(`https://wa.me/551140028922?text=${mensagem}`);
}

renderProdutos();
